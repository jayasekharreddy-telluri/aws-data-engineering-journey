import json
import boto3
from urllib.parse import unquote_plus

s3 = boto3.client('s3')   # init outside handler = warm-cached

def lambda_handler(event, context):
    for record in event['Records']:          # always loop — can be batched
        bucket = record['s3']['bucket']['name']
        raw_key = record['s3']['object']['key']
        key = unquote_plus(raw_key)          # decode URL-encoded spaces/specials

        print(f'Processing: s3://{bucket}/{key}')

        response = s3.get_object(Bucket=bucket, Key=key)
        content  = response['Body'].read().decode('utf-8')
        data     = json.loads(content)

        print(f'Menu items count: {len(data.get("items", []))}')

    return {'statusCode': 200, 'body': f'Processed {len(event["Records"])} records'}
