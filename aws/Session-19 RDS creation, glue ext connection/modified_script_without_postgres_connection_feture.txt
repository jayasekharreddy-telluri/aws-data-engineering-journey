import sys
import boto3
import json
from datetime import datetime

from awsglue.utils import getResolvedOptions
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from awsglue.job import Job
from awsglue.dynamicframe import DynamicFrame

from pyspark.sql import functions as F


# LOGGER
def log(message):
    print(f"[{datetime.now()}] {message}")


# READ JOB PARAMETERS
args = getResolvedOptions(sys.argv, [
    'JOB_NAME',
    'target_path',
    'secret_arn'
])

TARGET_PATH = args['target_path']
SECRET_ARN = args['secret_arn']


# GLUE CONTEXT INITIALIZATION
log("Initializing Glue Job")

sc = SparkContext()
glueContext = GlueContext(sc)
spark = glueContext.spark_session

job = Job(glueContext)
job.init(args['JOB_NAME'], args)

log(f"JOB NAME: {args['JOB_NAME']}")
log(f"TARGET PATH: {TARGET_PATH}")


# FETCH DB CREDENTIALS
log("Fetching credentials from Secrets Manager")

client = boto3.client("secretsmanager")

secret_response = client.get_secret_value(
    SecretId=SECRET_ARN
)

secret = json.loads(secret_response["SecretString"])

DB_USER = secret["username"]
DB_PASSWORD = secret["password"]


# JDBC CONFIGURATION
DB_HOST = "swiggy-customer-db.c5qo0eyiq2pe.ap-south-1.rds.amazonaws.com"
DB_PORT = "5432"
DB_NAME = "swiggy_external"
TABLE_NAME = "public.customer_feedback"

JDBC_URL = (
    f"jdbc:postgresql://{DB_HOST}:{DB_PORT}/{DB_NAME}"
)

log(f"JDBC URL: {JDBC_URL}")
log(f"SOURCE TABLE: {TABLE_NAME}")


# READ DATA FROM POSTGRES
log("Reading data from PostgreSQL")

feedback_dyf = glueContext.create_dynamic_frame.from_options(
    connection_type="postgresql",
    connection_options={
        "url": JDBC_URL,
        "dbtable": TABLE_NAME,
        "user": DB_USER,
        "password": DB_PASSWORD
    },
    transformation_ctx="postgres_read"
)

df = feedback_dyf.toDF()

log("PostgreSQL read successful")


# DEBUG SCHEMA
log("RAW SCHEMA")

df.printSchema()

log(f"RAW COLUMNS: {df.columns}")

raw_count = df.count()

log(f"RAW ROW COUNT: {raw_count}")


# VALIDATE DATA EXISTS
if raw_count == 0:
    raise Exception("No data found in source table")


# SAFE COLUMN NORMALIZATION
log("Normalizing column names")

clean_columns = []

for column in df.columns:

    cleaned_column = (
        column
        .replace(" ", "_")
        .replace("-", "_")
        .replace(".", "_")
        .strip()
        .lower()
    )

    clean_columns.append(cleaned_column)

log(f"CLEANED COLUMN NAMES: {clean_columns}")

df = df.toDF(*clean_columns)

log("Column normalization complete")

log(f"FINAL COLUMNS: {df.columns}")

df.printSchema()


# VALIDATE REQUIRED COLUMNS
required_columns = [
    "feedback_id",
    "customer_id",
    "order_id",
    "rating",
    "feedback_text",
    "sentiment",
    "feedback_date",
    "response_status"
]

missing_columns = []

for column in required_columns:
    if column not in df.columns:
        missing_columns.append(column)

if len(missing_columns) > 0:

    log(f"Missing columns: {missing_columns}")

    raise Exception(
        f"Required columns missing: {missing_columns}"
    )

log("All required columns exist")


# DATA CLEANING
log("Starting Silver Layer transformations")

cleaned_df = (
    df

    .withColumn(
        "sentiment",
        F.lower(
            F.trim(
                F.col("sentiment")
            )
        )
    )

    .withColumn(
        "feedback_text",
        F.trim(
            F.col("feedback_text")
        )
    )

    .filter(
        (F.col("rating") >= 1) &
        (F.col("rating") <= 5)
    )

    .filter(
        F.col("sentiment").isin(
            "positive",
            "neutral",
            "negative"
        )
    )
)

if "created_at" in cleaned_df.columns:

    cleaned_df = cleaned_df.drop("created_at")

    log("Dropped column: created_at")

log("Silver transformations completed")


# POST CLEAN VALIDATION
clean_count = cleaned_df.count()

log(f"CLEANED ROW COUNT: {clean_count}")

if clean_count == 0:
    raise Exception("No rows left after transformations")


# SENTIMENT DISTRIBUTION
log("SENTIMENT DISTRIBUTION")

cleaned_df.groupBy("sentiment").count().show()


# SAMPLE DATA
log("SAMPLE DATA")

cleaned_df.show(5, truncate=False)


# CONVERT TO DYNAMIC FRAME
log("Converting DataFrame to DynamicFrame")

final_dyf = DynamicFrame.fromDF(
    cleaned_df,
    glueContext,
    "final_dyf"
)


# WRITE TO S3
log("Writing Parquet files to S3")

glueContext.write_dynamic_frame.from_options(
    frame=final_dyf,
    connection_type="s3",
    connection_options={
        "path": TARGET_PATH,
        "partitionKeys": ["sentiment"]
    },
    format="parquet",
    transformation_ctx="s3_write"
)

log("Parquet write successful")


# JOB COMMIT
job.commit()


# SUCCESS
log("ETL JOB COMPLETED SUCCESSFULLY")

log("PIPELINE SUMMARY")
log(f"Source Database : {DB_NAME}")
log(f"Source Table    : {TABLE_NAME}")
log(f"Rows Processed  : {clean_count}")
log(f"Target Path     : {TARGET_PATH}")
log("Output Format   : parquet")
log("Partition Key   : sentiment")
